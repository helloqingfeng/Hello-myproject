第一章：目标站点可行性筛查（五步法）

在编写具体解析代码前，必须先探测目标站点的抗爬机制、架构模式与播放流真实有效性。

筛查维度 检查指标与特征 判定与应对策略
1. 网络与 WAF 防护 HTTP 403 / Cloudflare 5秒盾 / 状态码 850 / 海外阻断超时 放弃/不可用 纯静态 JS 扩展无法绕过 CF 5秒盾，直接跳过
2. CMS 架构识别 查看页面源码，确认是否为 MacCMS（苹果CMS）、海洋CMS等 推荐优选 MacCMS 规范度最高，自带 player_aaaa 或 player_data
3. 人机与弹窗限制 详情页/搜索页是否存在滑块验证（如 tncode） 部分受限 若仅搜索有防刷验证，可保留分类瀑布流，搜索做安全降级
4. 播放流结构逆向 检查播放页是否有直链 m3u8、Base64 加密、或二次签名接口 逆向分析 通过正则与解码逻辑提取出播放流直链
5. 视频流连通性测试 对提取的 .m3u8 发送 HEAD/GET 请求 校验通过 返回 HTTP 200 且包含 #EXTM3U 视为优质播放源

第二章：常见播放器加密与解密模式

在 getPlayinfo 中针对不同的播放器实现逆向还原：

模式 A：直链 m3u8 / mp4 最常见

无任何加密，数据直接在 player_data 对象的 url 属性中。

if (rawUrl.startsWith('http')) { return jsonify({ urls: [rawUrl] }); }
模式 B：MacCMS 标准加密 标准编码

声明 encrypt=1 (escape) 或 encrypt=2 (Base64 + escape)。

if (pObj.encrypt === 2) { rawUrl = unescape(atob(rawUrl)); } else if (pObj.encrypt === 1) { rawUrl = unescape(rawUrl); }
模式 C：二次签名接口 动态鉴权

如 YX/NBY 播放器，先请求签名接口获取带 token 链接再换取 m3u8。

const sRes = await $fetch.get( '?get_signed_url=1&url=' + encode(raw) ); const dRes = await $fetch.get(sRes.signed_url); return jsonify({ urls: [dRes.jmurl] });

第三章：XPTV 扩展标准代码规范（6 核心函数）

XPTV 运行于 JavaScript 沙箱环境，必须完整导出以下 6 个核心函数，且数据契约必须完全合规：

⚠️ 核心避坑铁律：
1. 入参转义：每个接收 ext 的函数开头必须执行 ext = argsify(ext)。
2. 线路与选集结构（重中之重）：getTracks 必须返回嵌套结构 { list: [ { title: '线路名', tracks: [ { name: '第1集', pan: '', ext: { url } } ] } ] }。若错误返回扁平数组，将导致客户端 “点击播放没反应”！
3. 播放结果返回：getPlayinfo 必须返回 { urls: [ streamUrl ] }。
标准 XPTV 扩展实现模版 (Template.js) JavaScript
// === 站点 XPTV 扩展脚本 (template.js) === const BASE_URL = 'https://www.example.com'; const HEADERS = { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36', 'Referer': BASE_URL + '/' }; // 1. 插件基础元信息 async function getLocalInfo() { return jsonify({ name: '示例影视', icon: BASE_URL + '/favicon.ico', version: '1.0.0', description: '海量超清影视免费在线播放', author: 'antigravity' }); } // 2. 顶部频道分类配置 async function getConfig() { return jsonify({ tabs: [ { name: '电影', ext: { url: BASE_URL + '/vodtype/1-{page}.html' } }, { name: '剧集', ext: { url: BASE_URL + '/vodtype/2-{page}.html' } }, { name: '动漫', ext: { url: BASE_URL + '/vodtype/3-{page}.html' } } ] }); } // 3. 瀑布流卡片解析 async function getCards(ext) { ext = argsify(ext); const page = ext.page || 1; let url = (ext.url || (BASE_URL + '/vodtype/1-{page}.html')).replace('{page}', page); try { const res = await $fetch.get(url, { headers: HEADERS }); const html = res.data; const cards = []; const regex = /<a[^>]+href=["'](\/voddetail\/\d+\.html)["'][^>]*>([\s\S]*?)<\/a>/gi; let match; while ((match = regex.exec(html)) !== null) { const rawA = match[0]; const link = BASE_URL + match[1].trim(); const titleMatch = rawA.match(/title=["']([^"']+)["']/i) || match[2].match(/alt=["']([^"']+)["']/i); const picMatch = rawA.match(/data-original=["']([^"']+)["']/i) || rawA.match(/src=["']([^"']+)["']/i); const remarksMatch = match[2].match(/<span class="[^"]*note[^"]*">([^<]*)<\/span>/i); const title = titleMatch ? titleMatch[1].trim() : ''; const pic = picMatch ? picMatch[1].trim() : ''; const remarks = remarksMatch ? remarksMatch[1].trim() : ''; if (title && !cards.find(c => c.ext.url === link)) { cards.push({ vod_id: link, vod_name: title, vod_pic: pic, vod_remarks: remarks, ext: { url: link } }); } } return jsonify({ list: cards }); } catch (e) { return jsonify({ list: [] }); } } // 4. 线路与选集提取 (严格遵循嵌套 list 结构) async function getTracks(ext) { ext = argsify(ext); const detailUrl = ext.url; if (!detailUrl) return jsonify({ list: [] }); try { const res = await $fetch.get(detailUrl, { headers: HEADERS }); const html = res.data; // 按播放线路 sid 分组提取 const epRegex = /href=["']\/vodplay\/(\d+)-(\d+)-(\d+)\.html["'][^>]*>([\s\S]*?)<\/a>/gi; let em; const lineMap = {}; while ((em = epRegex.exec(html)) !== null) { const [_, id, sid, nid] = em; const epName = em[4].replace(/<[^>]+>/g, '').trim() || `第${nid}集`; const epUrl = `${BASE_URL}/vodplay/${id}-${sid}-${nid}.html`; if (!lineMap[sid]) lineMap[sid] = []; if (!lineMap[sid].find(x => x.ext.url === epUrl)) { lineMap[sid].push({ name: epName, pan: '', ext: { url: epUrl } }); } } const lines = []; for (const sid of Object.keys(lineMap)) { lines.push({ title: `播放线路 ${sid}`, tracks: lineMap[sid] }); } return jsonify({ list: lines }); } catch (e) { return jsonify({ list: [] }); } } // 5. 播放流解析 async function getPlayinfo(ext) { ext = argsify(ext); const playUrl = ext.url; if (!playUrl) return jsonify({ urls: [] }); try { const res = await $fetch.get(playUrl, { headers: { ...HEADERS, 'Referer': playUrl } }); const html = res.data; const pMatch = html.match(/player_aaaa\s*=\s*({[^<]+?})</) || html.match(/player_data\s*=\s*({[^<]+?})</); if (pMatch) { const pObj = JSON.parse(pMatch[1]); let rawUrl = pObj.url; if (pObj.encrypt === 2) { try { rawUrl = unescape(atob(rawUrl)); } catch (err) {} } else if (pObj.encrypt === 1) { try { rawUrl = unescape(rawUrl); } catch (err) {} } if (rawUrl && (rawUrl.startsWith('http://') || rawUrl.startsWith('https://'))) { return jsonify({ urls: [rawUrl] }); } } const m3u8Match = html.match(/https?:\/\/[^\s"'<>]+\.m3u8[^\s"'<>]*/); if (m3u8Match) { return jsonify({ urls: [m3u8Match[0]] }); } return jsonify({ urls: [] }); } catch (e) { return jsonify({ urls: [] }); } } // 6. 搜索支持 async function search(ext) { ext = argsify(ext); const keyword = ext.text || ''; if (!keyword) return jsonify({ list: [] }); try { const searchUrl = `${BASE_URL}/vodsearch/${encodeURIComponent(keyword)}-------------.html`; const res = await $fetch.get(searchUrl, { headers: HEADERS }); // 解析卡片列表... return jsonify({ list: [] }); } catch (e) { return jsonify({ list: [] }); } }

第四章：Node.js 沙箱端到端仿真测试

无需将脚本传到手机或 Apple TV 即可在本地全自动验证所有 6 个函数与视频直链连通性：

仿真测试脚本 (simulate_test.js) Node.js VM
const fs = require('fs'); const vm = require('vm'); async function testExtension(filePath) { const code = fs.readFileSync(filePath, 'utf8'); // 模拟 XPTV 原生沙箱上下文 const sandbox = { jsonify: (obj) => JSON.stringify(obj), argsify: (str) => typeof str === 'string' ? JSON.parse(str) : (str || {}), atob: (str) => Buffer.from(str, 'base64').toString('binary'), btoa: (str) => Buffer.from(str, 'binary').toString('base64'), escape: escape, unescape: unescape, console: console, $fetch: { get: async (url, opts) => { const res = await fetch(url, { headers: opts?.headers || {}, signal: AbortSignal.timeout(10000) }); return { data: await res.text(), status: res.status }; } } }; const context = vm.createContext(sandbox); vm.runInContext(code, context); console.log('1. getLocalInfo:', JSON.parse(await vm.runInContext('getLocalInfo()', context))); const config = JSON.parse(await vm.runInContext('getConfig()', context)); console.log('2. getConfig 分类数:', config.tabs.length); const cardsRes = JSON.parse(await vm.runInContext(`getCards(${JSON.stringify(JSON.stringify(config.tabs[0].ext))})`, context)); console.log(`3. getCards 成功获取卡片: ${cardsRes.list.length} 条`); if (cardsRes.list.length > 0) { const tracksRes = JSON.parse(await vm.runInContext(`getTracks(${JSON.stringify(JSON.stringify(cardsRes.list[0].ext))})`, context)); console.log(`4. getTracks 解析线路数: ${tracksRes.list.length} 条`); if (tracksRes.list.length > 0 && tracksRes.list[0].tracks.length > 0) { const playRes = JSON.parse(await vm.runInContext(`getPlayinfo(${JSON.stringify(JSON.stringify(tracksRes.list[0].tracks[0].ext))})`, context)); console.log('5. getPlayinfo 播放地址:', playRes); if (playRes.urls?.[0]) { const streamCheck = await fetch(playRes.urls[0], { method: 'HEAD' }); console.log(`6. 视频流连通性测试: HTTP ${streamCheck.status} ✅`); } } } } testExtension('./kxyy.js');
